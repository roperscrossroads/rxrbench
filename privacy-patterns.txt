# Generic site-private patterns (tools/privacy_check.py). A site's own list is
# <site>/private-patterns.txt. One regular expression per line, case-insensitive.

# Private IPv4 ranges (RFC 1918). Examples use the documentation ranges (192.0.2.0/24 etc.).
\b10\.\d{1,3}\.\d{1,3}\.\d{1,3}\b
\b172\.(1[6-9]|2\d|3[01])\.\d{1,3}\.\d{1,3}\b
\b192\.168\.\d{1,3}\.\d{1,3}\b
# Internal DNS names
\b[a-z0-9-]+\.home\.arpa\b
\b[a-z0-9-]+\.lan\b
# A real user's home directory (examples use ~ or /home/bench)
/home/(?!bench\b)[a-z_][a-z0-9_-]*/
# Email addresses, except the noreply kind
\b[a-z0-9._%+-]+@(?!example\.)(?!users\.noreply\.)[a-z0-9.-]+\.[a-z]{2,}\b
# mDNS names (examples use benchpi<n>.local)
\b(?!claude\.|benchpi)[a-z0-9-]+\.local\b
# IPv6 unique local addresses
\bf[cd][0-9a-f]{2}:[0-9a-f:]+
# A real user's home directory on macOS or Windows
/Users/(?!bench\b)[a-z_][a-z0-9_-]*/
[a-z]:\\users\\
